#启动、查看、结束进程的windows命令
#查看
# 查找所有包含 uvicorn 的进程
tasklist | findstr uvicorn
# 查看 8000 端口被哪个进程占用
netstat -ano | findstr :8000
# 查找 Node.js 进程
tasklist | findstr node.exe
# 查看 5173 端口
netstat -ano | findstr :5173

# 查看 8080 端口
netstat -ano | findstr :8080

#结束
# 强制结束 PID 为 12345 的进程
taskkill /F /PID 12345
# 结束所有 uvicorn 相关的 Python 进程（谨慎使用）
taskkill /F /IM python.exe

# 结束所有 Node.js 进程（谨慎！会关闭所有 Node 应用）
taskkill /F /IM node.exe
#启动
# 开发模式 (默认端口 8080 或 5173)
# 启动前端
cd C:\MyProject\project8_bug\web\frontend
npm run dev

# 启动后端
cd C:\MyProject\project8_bug\web\backend
conda activate tick_detection
uvicorn app.main:app --reload --port 8000
